--------------database----------
CREATE DATABASE bank_analysis;
USE bank_analysis;

------table-------
CREATE TABLE bank_customers (
    RowNumber INT,
    CustomerId INT PRIMARY KEY,
    Surname VARCHAR(50),
    CreditScore INT,
    Geography VARCHAR(20),
    Gender VARCHAR(10),
    Age INT,
    Tenure INT,
    Balance DECIMAL(12,2),
    NumOfProducts INT,
    HasCrCard TINYINT,
    IsActiveMember TINYINT,
    EstimatedSalary DECIMAL(12,2),
    Exited TINYINT
);
